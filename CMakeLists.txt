# The Apinator regions (game/net/net_apinator.cpp): every
# tools/netserver/apinator-<region>.env (APINATOR_KEY, APINATOR_CLUSTER, and
# APINATOR_SECRET or APINATOR_AUTH_URL; optional APINATOR_LABEL, APINATOR_URL)
# becomes one entry of a generated header. With an auth URL the secret is
# left out of the build. A secret that is built in can be read out of the
# game by anyone who has it (a web build especially), so a region with no
# auth URL is skipped unless CH_APINATOR_EMBED_SECRET is on (private builds).
# A region without a local .env (a fresh clone, CI) takes its committed
# apinator-<region>.public instead: key, cluster and auth URL only; a secret
# in a .public file is never read.
option(CH_APINATOR_EMBED_SECRET "Build an Apinator app secret into the game (never for builds you share)" OFF)
file(GLOB CH_APINATOR_ENVS ${CMAKE_SOURCE_DIR}/tools/netserver/apinator-*.env)
file(GLOB CH_APINATOR_PUBLIC ${CMAKE_SOURCE_DIR}/tools/netserver/apinator-*.public)
foreach(pub ${CH_APINATOR_PUBLIC})
  string(REGEX REPLACE "\\.public$" ".env" env "${pub}")
  if(NOT EXISTS "${env}")
    list(APPEND CH_APINATOR_ENVS "${pub}")
  endif()
endforeach()
list(SORT CH_APINATOR_ENVS ORDER DESCENDING)   # us, eu: the first is the default
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CH_APINATOR_PUBLIC})
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CH_APINATOR_ENVS})
set(CH_APINATOR_ROWS "")
set(CH_APINATOR_COUNT 0)
foreach(env ${CH_APINATOR_ENVS})
  file(STRINGS ${env} lines REGEX "^APINATOR_[A-Z_]+=")
  foreach(v KEY SECRET CLUSTER AUTH_URL LABEL URL)
    set(A_${v} "")
  endforeach()
  foreach(line ${lines})
    string(REGEX MATCH "^APINATOR_([A-Z_]+)=(.*)$" _ "${line}")
    string(STRIP "${CMAKE_MATCH_2}" val)
    set(A_${CMAKE_MATCH_1} "${val}")
  endforeach()
  if(env MATCHES "\\.public$")
    set(A_SECRET "")
  endif()
  if(A_KEY STREQUAL "" OR A_CLUSTER STREQUAL "")
    continue()
  endif()
  if(NOT A_AUTH_URL STREQUAL "")
    set(A_SECRET "")
  elseif(NOT CH_APINATOR_EMBED_SECRET)
    get_filename_component(name ${env} NAME)
    message(STATUS "Apinator: ${name} has no APINATOR_AUTH_URL; left out (CH_APINATOR_EMBED_SECRET is off)")
    continue()
  endif()
  if(A_LABEL STREQUAL "")
    string(TOUPPER "${A_CLUSTER}" up)
    set(A_LABEL "Apinator ${up}")
  endif()
  if(A_URL STREQUAL "")
    set(A_URL "wss://ws-${A_CLUSTER}.apinator.io")
  endif()
  string(APPEND CH_APINATOR_ROWS "    {\"${A_CLUSTER}\", \"${A_LABEL}\", \"${A_URL}\", \"${A_KEY}\", \"${A_SECRET}\", \"${A_AUTH_URL}\"},\n")
  math(EXPR CH_APINATOR_COUNT "${CH_APINATOR_COUNT} + 1")
endforeach()
if(CH_APINATOR_COUNT EQUAL 0)
  set(CH_APINATOR_ROWS "    {nullptr, nullptr, nullptr, nullptr, nullptr, nullptr},\n")
endif()
file(CONFIGURE OUTPUT ${CMAKE_BINARY_DIR}/generated/apinator_config.gen.hpp CONTENT
"// Generated by CMakeLists.txt from tools/netserver/apinator-*.env: do not edit.
#pragma once
namespace ch::net::apinator_config {
struct Region { const char* region; const char* label; const char* url; const char* key; const char* secret; const char* auth; };
inline constexpr Region kRegions[] = {
${CH_APINATOR_ROWS}};
inline constexpr int kRegionCount = ${CH_APINATOR_COUNT};
}
" @ONLY)

if(EMSCRIPTEN)

  set(CH_EM_PORTS -sUSE_SDL=2 -sUSE_SDL_TTF=2 -sUSE_ZLIB=1 -fexceptions)

    list(APPEND CH_EM_PORTS -sUSE_SDL_MIXER=2 "-sSDL2_MIXER_FORMATS=[\"mp3\",\"wav\"]")
  endif()

  target_link_options(game PRIVATE ${CH_EM_PORTS} -sALLOW_MEMORY_GROWTH=1 -lidbfs.js -lwebsocket.js -sFETCH=1
                      "--preload-file=${CMAKE_SOURCE_DIR}/resources@/resources"
                      --shell-file ${CMAKE_SOURCE_DIR}/web/shell.html
                      "-sEXPORTED_RUNTIME_METHODS=[\"ccall\",\"FS\",\"addRunDependency\",\"removeRunDependency\"]"
                      -sFORCE_FILESYSTEM=1)
  set_property(TARGET game APPEND PROPERTY LINK_DEPENDS ${CMAKE_SOURCE_DIR}/web/shell.html)
  # Distro Emscripten packages (Debian, Ubuntu) leave out html-minifier, which
  # an optimised link needs; -g2 skips that step (the page is small anyway).
  if(NOT EXISTS "${EMSCRIPTEN_ROOT_PATH}/node_modules/.bin/html-minifier-terser")
    target_link_options(game PRIVATE -g2)
  endif()
else()

    target_link_libraries(game PRIVATE OpenSSL::SSL OpenSSL::Crypto)
  endif()
  find_package(ZLIB REQUIRED)
  target_link_libraries(game PRIVATE SDL2::SDL2 SDL2::SDL2main SDL2_ttf::SDL2_ttf ZLIB::ZLIB)
  if(WIN32)
    target_link_libraries(game PRIVATE ws2_32)     # Winsock (game/net/net_transport.cpp)
  endif()
  if(CH_AUDIO)
    find_package(SDL2_mixer REQUIRED)
    target_link_libraries(game PRIVATE SDL2_mixer::SDL2_mixer)
  endif()

  else()
